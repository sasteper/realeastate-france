11def validate_surface(value: str) -> tuple[bool, Optional[float], str]:
12    """
13    Validate property surface area input.
14
15    Args:
16        value: Raw string from the input field.
17
18    Returns:
19        Tuple of (is_valid, parsed_float_or_None, error_message).
20    """
21    value = value.strip().replace(",", ".")
22    if not value:
23        return False, None, "Pole wymagane"
24    try:
25        v = float(value)
26    except ValueError:
27        return False, None, "Podaj liczbę (np. 85)"
28    if v <= 0:
29        return False, None, "Powierzchnia musi być > 0"
30    if v > 5000:
31        return False, None, "Max 5 000 m²"
32    return True, v, ""

 82    def save_estimation(
 83        self,
 84        surface_m2: float,
 85        bedrooms: int,
 86        dpe_class: str,
 87        image_count: int,
 88        property_type: str,
 89        amenities: dict,
 90        model_used: str,

 93        price_high: float,
 94        valuation_label: Optional[str] = None,
 95    ) -> int:
 96        """
 97        Persist a price estimation to the database.
 98
 99        Args:
100            surface_m2: Property surface area in m².
101            bedrooms: Number of bedrooms.
102            dpe_class: Energy class (A–G).
103            image_count: Number of listing images.
104            property_type: 'apartment', 'house', or 'land'.
105            amenities: Dict of boolean amenity flags.
106            model_used: 'OLS' or 'RandomForest'.
107            price_estimate: Point estimate in EUR.
108            price_low: Lower bound of 95% prediction interval.
109            price_high: Upper bound of 95% prediction interval.
110            valuation_label: 'Okazyjna', 'Przeciętna', or 'Za wysoka'.
111
112        Returns:
113            Row ID of the inserted record.
114        """
115        ts = datetime.now().isoformat(timespec="seconds")
116        with self._get_connection() as conn:
117            cur = conn.execute(
118                """
119                INSERT INTO estimations (
120                    timestamp, surface_m2, bedrooms, dpe_class, image_count,
121                    property_type, has_pool, has_terrace, has_view, has_garden,
122                    has_transport, is_quiet, has_fireplace, needs_work,
123                    model_used, price_estimate, price_low, price_high, valuation_label
124                ) VALUES (?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?)
125                """,
126                (
127                    ts, surface_m2, bedrooms, dpe_class, image_count,
128                    property_type,
129                    int(amenities.get("has_pool", 0)),
130                    int(amenities.get("has_terrace", 0)),
131                    int(amenities.get("has_view", 0)),
132                    int(amenities.get("has_garden", 0)),
133                    int(amenities.get("has_transport", 0)),
134                    int(amenities.get("is_quiet", 0)),
135                    int(amenities.get("has_fireplace", 0)),
136                    int(amenities.get("needs_work", 0)),
137                    model_used, price_estimate, price_low, price_high,
138                    valuation_label,
139                ),
140            )
141            row_id = cur.lastrowid
142        self.log_action("ESTIMATION", f"Saved estimation id={row_id}, model={model_used}, price={price_estimate:.0f}")
143        return row_id

159    def predict(
160        self,
161        surface_m2: float,
162        bedrooms: int,
163        dpe_class: str,
164        image_count: int,
165        property_type: Literal["apartment", "house", "land"],
166        amenities: dict[str, int],
167        model: Literal["OLS", "RandomForest"] = "OLS",
168    ) -> PredictionResult:
169        """
170        Predict property price and compute a 95% prediction interval.
171
172        Args:
173            surface_m2: Property area in m² (must be > 0).
174            bedrooms: Number of bedrooms (≥ 0).
175            dpe_class: DPE energy class ('A'–'G').
176            image_count: Number of listing images (3–13).
177            property_type: 'apartment', 'house', or 'land'.
178            amenities: Dict of boolean feature flags (0 or 1).
179            model: Which model to use — 'OLS' or 'RandomForest'.
180
181        Returns:
182            :class:`PredictionResult` with point estimate, interval, and label.
183
184        Raises:
185            ValueError: If ``surface_m2`` ≤ 0 or ``dpe_class`` is invalid.
186        """
187        if surface_m2 <= 0:
188            raise ValueError(f"surface_m2 must be positive, got {surface_m2}")
189        if dpe_class not in DPE_MAP:
190            raise ValueError(f"Invalid DPE class: {dpe_class!r}")
191
192        X = self._build_feature_vector(
193            surface_m2, bedrooms, dpe_class, image_count, property_type, amenities
194        )
195
196        if model == "OLS":
197            Xc = sm.add_constant(X, has_constant="add")
198            log_pred = float(self._ols.predict(Xc).iloc[0])
199            std = self._ols_std
200            name = "OLS"
201        else:
202            log_pred = float(self._rf.predict(X)[0])
203            std = self._rf_std
204            name = "RandomForest"
205
206        log_pred = self._clamp_log_price(log_pred)
207        log_low = self._clamp_log_price(log_pred - 1.96 * std)
208        log_high = self._clamp_log_price(log_pred + 1.96 * std)
209
210        price = float(np.expm1(log_pred))
211        price_low = float(np.expm1(log_low))
212        price_high = float(np.expm1(log_high))
213
214        # Round to nearest 500 EUR for cleaner display
215        price = round(price / 500) * 500
216        price_low = round(price_low / 500) * 500
217        price_high = round(price_high / 500) * 500
218
219        label = self._valuation_label(price, self._price_p25, self._price_p75)
220
221        logger.debug(
222            "Prediction [%s]: %.0f EUR [%.0f – %.0f] — %s",
223            name, price, price_low, price_high, label,
224        )
225
226        return PredictionResult(
227            model_name=name,
228            price_estimate=price,
229            price_low=price_low,
230            price_high=price_high,
231            valuation_label=label,
232            log_price_estimate=log_pred,
233        )
